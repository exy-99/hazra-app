// Row wiring (extends 03-03 applyStatus — same pendingIds set, new note path)
async function saveNote(workerId: string, text: string | null) {
  const row = rows.find((r) => r.worker_id === workerId);
  if (!row || pendingIds.has(workerId)) return;
  // Unmarked + note edge: a note NEEDS a status row — default to 'present' when row.status == null
  const status = row.status ?? 'present';
  await upsertAttendance({ workerId, date: selectedDate, status, note: text });
}
