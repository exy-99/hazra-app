const [pendingIds, setPendingIds] = useState<Set<string>>(new Set());
async function applyStatus(workerId: string, next: AttendanceStatus) {
  if (pendingIds.has(workerId)) return;                 // second tap of a rapid double-tap is dropped
  setPendingIds((p) => new Set(p).add(workerId));
  try { await upsertAttendance({ workerId, date: selectedDate, status: next, note: currentNote(workerId) /* 03-04 owns notes; here preserve loaded note */ }); setRows(...); }
  catch { setSaveError(true); } finally { setPendingIds(...delete...); }
}

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-07 | Tampering | duplicate (worker,date) rows under race | mitigate | Single-statement `ON CONFLICT(worker_id,date) DO UPDATE` (DB-level dedupe) + per-row `pendingIds` handler guard; no read-then-write split |
| T-03-08 | Tampering | worker_id interpolation | mitigate | `workerId` binds as a `?` parameter inside the DAO; screen never concatenates SQL (DELETE grep-gate 0) |
| T-03-09 | Denial of service | write failure bricks the register | mitigate | Failure sets a polite banner only; rows stay interactive; retry is a re-tap; `finally` always clears pending |
| T-03-10 | Information disclosure | soft-deleted workers reappear | mitigate | `getAttendanceForDate` filters `is_active = 1` server-side (DAO); screen never passes `includeInactive` |
</threat_model>
